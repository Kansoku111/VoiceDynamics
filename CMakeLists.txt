cmake_minimum_required(VERSION 3.25)
project(VoiceDynamics VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "Statically link the Microsoft runtime" FORCE)
set(VST3_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vst3sdk" CACHE PATH "Steinberg VST3 SDK")
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)

# DSP, measurement and analysis: no SDK, no windowing code.
add_library(vd_dsp STATIC src/loudness.cpp src/truepeak.cpp src/dynamics.cpp src/limiter.cpp src/chain.cpp src/analyzer.cpp src/learner.cpp src/wavio.cpp)
target_include_directories(vd_dsp PUBLIC src)
target_compile_definitions(vd_dsp PUBLIC NOMINMAX _USE_MATH_DEFINES)

add_executable(vd_conformance_tests tests/conformance_tests.cpp)
target_link_libraries(vd_conformance_tests PRIVATE vd_dsp)
target_include_directories(vd_conformance_tests PRIVATE tests)
add_executable(vd_dsp_tests tests/dsp_tests.cpp)
target_link_libraries(vd_dsp_tests PRIVATE vd_dsp)
target_include_directories(vd_dsp_tests PRIVATE tests)
add_executable(vd_tool tools/vd_tool.cpp)
target_link_libraries(vd_tool PRIVATE vd_dsp)
target_include_directories(vd_tool PRIVATE tests)

set(VD_TARGETS vd_dsp vd_conformance_tests vd_dsp_tests vd_tool)
set(VD_EXECUTABLES vd_conformance_tests vd_dsp_tests vd_tool)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/plugin.cpp")
  smtg_add_vst3plugin(VoiceDynamics src/plugin.cpp src/editor.cpp)
  target_include_directories(VoiceDynamics PRIVATE "${VST3_SDK_ROOT}")
  target_link_libraries(VoiceDynamics PRIVATE sdk vd_dsp comctl32 user32 gdi32 gdiplus shell32 comdlg32)
  add_executable(vd_host_tests tests/host_tests.cpp "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_include_directories(vd_host_tests PRIVATE "${VST3_SDK_ROOT}")
  target_link_libraries(vd_host_tests PRIVATE sdk_hosting sdk vd_dsp comctl32 user32 gdi32)
  add_executable(vd_editor_tests tests/editor_tests.cpp "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_include_directories(vd_editor_tests PRIVATE "${VST3_SDK_ROOT}")
  target_link_libraries(vd_editor_tests PRIVATE sdk_hosting sdk vd_dsp comctl32 user32 gdi32 gdiplus)
  list(APPEND VD_TARGETS VoiceDynamics vd_host_tests vd_editor_tests)
  list(APPEND VD_EXECUTABLES vd_host_tests vd_editor_tests)
endif()

enable_testing()
add_test(NAME loudness_conformance COMMAND vd_conformance_tests)
add_test(NAME dsp_behavior COMMAND vd_dsp_tests)
if(TARGET VoiceDynamics)
  add_test(NAME vst3_host COMMAND vd_host_tests $<TARGET_FILE:VoiceDynamics>)
  add_test(NAME vst3_editor COMMAND vd_editor_tests $<TARGET_FILE:VoiceDynamics>)
endif()
if(MSVC)
  foreach(t ${VD_TARGETS})
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
  endforeach()
  foreach(t moduleinfotool validator ${VD_EXECUTABLES})
    if(TARGET ${t})
      target_link_options(${t} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/utf8.manifest")
    endif()
  endforeach()
endif()
